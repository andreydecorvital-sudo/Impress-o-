function pairPages(items){const shipping=[],details=[];for(const item of items){const ss=shippingScore(item.text),ds=detailScore(item.text);item.keys=orderKeys(item.text);if(ss>=ds&&ss>=2)shipping.push(item);else details.push(item)}if(!shipping.length&&items.length){shipping.push(items[0]);details.push(...items.slice(1))}const used=new Set(),pairs=[];for(const ship of shipping){let best=null,bestScore=-1e9;for(const det of details){if(used.has(det.index))continue;const shared=ship.keys.some(k=>det.keys.includes(k)),distance=Math.abs(det.index-ship.index);let score=(shared?100:0)-distance;if(det.index>ship.index)score+=4;if(score>bestScore){best=det;bestScore=score}}if(best)used.add(best.index);pairs.push({ship,detail:best})}return pairs}
function rotateCanvas(src,turns){turns=((turns%4)+4)%4;if(!turns)return src;const out=document.createElement('canvas'),swap=turns%2===1;out.width=swap?src.height:src.width;out.height=swap?src.width:src.height;const ctx=out.getContext('2d');ctx.translate(out.width/2,out.height/2);ctx.rotate(turns*Math.PI/2);ctx.drawImage(src,-src.width/2,-src.height/2);return out}
function inkBounds(canvas,threshold=245,margin=8){const ctx=canvas.getContext('2d'),data=ctx.getImageData(0,0,canvas.width,canvas.height).data;let minX=canvas.width,minY=canvas.height,maxX=-1,maxY=-1;const step=2;for(let y=0;y<canvas.height;y+=step)for(let x=0;x<canvas.width;x+=step){const p=(y*canvas.width+x)*4;if(data[p]<threshold||data[p+1]<threshold||data[p+2]<threshold){if(x<minX)minX=x;if(y<minY)minY=y;if(x>maxX)maxX=x;if(y>maxY)maxY=y}}if(maxX<0)return{x:0,y:0,w:canvas.width,h:canvas.height};minX=Math.max(0,minX-margin);minY=Math.max(0,minY-margin);maxX=Math.min(canvas.width-1,maxX+margin);maxY=Math.min(canvas.height-1,maxY+margin);return{x:minX,y:minY,w:maxX-minX+1,h:maxY-minY+1}}
function cropCanvas(src,b){const out=document.createElement('canvas');out.width=Math.max(1,b.w);out.height=Math.max(1,b.h);out.getContext('2d').drawImage(src,b.x,b.y,b.w,b.h,0,0,b.w,b.h);return out}
function prepareDetail(item){let src=item.canvas;const b0=inkBounds(src);if(b0.h>b0.w*1.18)src=rotateCanvas(src,1);let cropped=cropCanvas(src,inkBounds(src));if(cropped.height>cropped.width*1.35)cropped=rotateCanvas(cropped,1);return cropCanvas(cropped,inkBounds(cropped))}
function occupancy(canvas,cols=34,rows=50){const ctx=canvas.getContext('2d'),data=ctx.getImageData(0,0,canvas.width,canvas.height).data,grid=[];for(let gy=0;gy<rows;gy++){const row=[];for(let gx=0;gx<cols;gx++){const x0=Math.floor(gx*canvas.width/cols),x1=Math.floor((gx+1)*canvas.width/cols),y0=Math.floor(gy*canvas.height/rows),y1=Math.floor((gy+1)*canvas.height/rows);let dark=0,total=0;const sx=Math.max(1,Math.floor((x1-x0)/6)),sy=Math.max(1,Math.floor((y1-y0)/6));for(let y=y0;y<y1;y+=sy)for(let x=x0;x<x1;x+=sx){const p=(y*canvas.width+x)*4;total++;if(data[p]<225)dark++}row.push(dark/Math.max(1,total)>.02?1:0)}grid.push(row)}return grid}
function largestEmptyRect(canvas){const cols=34,rows=50,g=occupancy(canvas,cols,rows),heights=new Array(cols).fill(0);let best={area:0,x:0,y:0,w:0,h:0};for(let r=0;r<rows;r++){for(let c=0;c<cols;c++)heights[c]=g[r][c]?0:heights[c]+1;const stack=[];for(let c=0;c<=cols;c++){const cur=c===cols?0:heights[c];while(stack.length&&heights[stack[stack.length-1]]>cur){const top=stack.pop(),h=heights[top],left=stack.length?stack[stack.length-1]+1:0,w=c-left,area=w*h;if(area>best.area)best={area,x:left,y:r-h+1,w,h}}stack.push(c)}}return{x:Math.floor(best.x*canvas.width/cols),y:Math.floor(best.y*canvas.height/rows),w:Math.floor(best.w*canvas.width/cols),h:Math.floor(best.h*canvas.height/rows),ratio:best.area/(cols*rows)}}
function fitImage(ctx,src,box,pad=8){const aw=Math.max(1,box.w-pad*2),ah=Math.max(1,box.h-pad*2),scale=Math.min(aw/src.width,ah/src.height),w=Math.max(1,Math.floor(src.width*scale)),h=Math.max(1,Math.floor(src.height*scale)),x=Math.floor(box.x+(box.w-w)/2),y=Math.floor(box.y+(box.h-h)/2);ctx.fillStyle='#fff';ctx.fillRect(box.x,box.y,box.w,box.h);ctx.drawImage(src,0,0,src.width,src.height,x,y,w,h)}
