function fuse(shipCanvas,detailCanvas){const rect=largestEmptyRect(shipCanvas),minW=shipCanvas.width*.55,minH=shipCanvas.height*.16,useInside=rect.ratio>.10&&rect.w>=minW&&rect.h>=minH;const extraH=useInside?0:Math.max(250,Math.round(shipCanvas.width*.34)),out=document.createElement('canvas');out.width=shipCanvas.width;out.height=shipCanvas.height+extraH;const ctx=out.getContext('2d');ctx.fillStyle='#fff';ctx.fillRect(0,0,out.width,out.height);ctx.drawImage(shipCanvas,0,0);const box=useInside?rect:{x:8,y:shipCanvas.height+8,w:out.width-16,h:extraH-16};ctx.fillStyle='#fff';ctx.fillRect(box.x,box.y,box.w,box.h);ctx.strokeStyle='#000';ctx.lineWidth=Math.max(3,Math.round(out.width/250));ctx.strokeRect(box.x+2,box.y+2,box.w-4,box.h-4);const titleH=Math.max(30,Math.round(out.width/25));ctx.fillStyle='#000';ctx.font='bold '+Math.max(20,Math.round(out.width/34))+'px Arial';ctx.fillText('SEPARAÇÃO / ITENS DO PEDIDO',box.x+12,box.y+titleH);fitImage(ctx,detailCanvas,{x:box.x+6,y:box.y+titleH+5,w:box.w-12,h:box.h-titleH-11},4);return{canvas:out,inside:useInside}}
