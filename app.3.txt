function cleanPackingText(s){return String(s||'').replace(/[|_]+/g,' ').replace(/\s+/g,' ').trim()}
function packingQuantity(text){const t=cleanPackingText(text);let orderQty=Number((t.match(/(?:qtd|quantidade|qtde)\s*[:#-]?\s*(\d+)/i)||[])[1]||1),m;
 const patterns=[/(\d+)\s*(?:kits?|pacotes?|caixas?|conjuntos?)\s*(?:de|com|c\/)?\s*(\d+)/i,/(\d+)\s*[xX]\s*(\d+)/i];
 for(const rx of patterns){m=t.match(rx);if(m)return Number(m[1])*Number(m[2])}
 m=t.match(/(?:kit|pacote|caixa|conjunto)\s*(?:de|com|c\/)?\s*(\d+)/i);if(m)return orderQty*Number(m[1]);
 return orderQty}

 if(!line||line.length<3)line=lines.filter(x=>x.length>=3&&!blocked.test(x)&&!/^\d+[,.]?\d*$/.test(x)).sort((a,b)=>b.length-a.length)[0]||'';
 line=line.replace(/\b(?:kit|pacote|caixa|conjunto)\s*(?:de|com|c\/)?\s*\d+\b/ig,' ').replace(/\b\d+\s*[xX]\s*\d+\b/g,' ').replace(/\b(?:unidades?|unid\.?|pcs?|pe[cç]as?)\b/ig,' ').replace(/\b(?:para|com|de|da|do|das|dos|em|e|a|o)\b/ig,' ').replace(/\s+/g,' ').trim();

function packingLabel(text){return packingQuantity(text)+' '+packingProduct(text)}
